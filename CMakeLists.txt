cmake_minimum_required(VERSION 3.28)
project(arconaut LANGUAGES CXX)

set(ARCONAUT_COMPILER_PROFILE "macos-llvm23" CACHE STRING "Explicit qualified compiler profile")
if(ARCONAUT_COMPILER_PROFILE STREQUAL "macos-llvm23")
  set(ARCONAUT_REQUIRED_CLANG "23.1.2")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    message(FATAL_ERROR "macos-llvm23 requires its declared Darwin host")
  endif()
elseif(ARCONAUT_COMPILER_PROFILE STREQUAL "linux-clang18")
  set(ARCONAUT_REQUIRED_CLANG "18.1.3")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "linux-clang18 requires its declared Linux host")
  endif()
else()
  message(FATAL_ERROR "Unknown compiler profile; refusing fallback")
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "${ARCONAUT_REQUIRED_CLANG}")
  message(FATAL_ERROR "This tooling profile requires LLVM Clang ${ARCONAUT_REQUIRED_CLANG}")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF) # No C++ modules selected in this owned foundation.
set(ARCONAUT_INSTRUMENTATION "none" CACHE STRING "none, address, thread, fuzz")
set_property(CACHE ARCONAUT_INSTRUMENTATION PROPERTY STRINGS none address thread fuzz)

add_library(arconaut_build INTERFACE)
if(ARCONAUT_COMPILER_PROFILE STREQUAL "linux-clang18")
  # Ubuntu's installed libstdc++ 13 supports the C++20 thread contract.
  target_compile_options(arconaut_build INTERFACE -stdlib=libstdc++)
  target_link_options(arconaut_build INTERFACE -stdlib=libstdc++)
endif()
target_compile_options(arconaut_build INTERFACE
  -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion -Wshadow
  -Wnon-virtual-dtor -Wold-style-cast -Woverloaded-virtual -Wformat=2
  -Wundef -Wimplicit-fallthrough -Werror=unknown-warning-option)
if(ARCONAUT_INSTRUMENTATION STREQUAL "address")
  target_compile_options(arconaut_build INTERFACE
    -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(arconaut_build INTERFACE -fsanitize=address,undefined)
elseif(ARCONAUT_INSTRUMENTATION STREQUAL "thread")
  target_compile_options(arconaut_build INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(arconaut_build INTERFACE -fsanitize=thread)
elseif(ARCONAUT_INSTRUMENTATION STREQUAL "fuzz")
  target_compile_options(arconaut_build INTERFACE
    -fsanitize=fuzzer-no-link,address,undefined -fno-sanitize-recover=all)
  target_link_options(arconaut_build INTERFACE -fsanitize=address,undefined)
elseif(NOT ARCONAUT_INSTRUMENTATION STREQUAL "none")
  message(FATAL_ERROR "Unknown instrumentation; refusing fallback")
endif()
